    /// <summary>Checks exact values, primitive numeric types, signed zero and bounded overflow.</summary>
    /// <param name="profile">The XPath compatibility profile.</param>
    /// <param name="expression">The oracle predicate.</param>

    [Theory, MemberData(nameof(Cases))]
    public void ExactKernel(XPathCompatibility profile, string expression, bool error)
    {
        string Evaluate() => XPath31Expression.Compile(expression, new CompileOptions { Compatibility = profile }).Evaluate(new EvaluationContext()).ToString();
        if (error) Assert.Contains("FOAR0002", Assert.ThrowsAny<Exception>(() => Evaluate()).Message);
        else Assert.Equal("true", Evaluate());
    }
