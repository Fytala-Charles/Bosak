from pathlib import Path
import json,shutil,re
C=Path('docs/xpath40-catalog');E=C/'evidence/c12-semantic-batch4'
def load(p):return json.loads(p.read_text(encoding='utf-8-sig'))
def save(n,v):(E/n).write_text(json.dumps(v,indent=2,ensure_ascii=False)+'\n',encoding='utf-8')
q=[]
for v in load(C/'evidence/c05-collection-pattern-review/probe-requests.json'):
 if v['row'] not in ['S6-types','S6-record-conversion','S6-update']:continue
 v=dict(v,projection=None)
 v['entry']='type-system/'+('enum-types.md' if v['id'].startswith('enum') else 'choice-types.md' if v['id'].startswith('choice') else 'record-updates.md' if v['row']=='S6-update' else 'typed-let-coercion.md' if 'typed-let' in v['id'] else 'structural-records.md')
 q.append(v)
save('probe-requests.json',q)
h=[]
for v in load(C/'evidence/c06-host-profile-review/host-requests.json'):
 if v['row']=='S0' and not v['id'].startswith(('local-','serialization-','scan-','future-')):continue
 id=v['id'];page='processor-profiles' if v['row']=='S0' else 'xsl-note' if id.startswith('note') else 'xsl-if' if id.startswith('if-') else 'xsl-map' if id.startswith('map-') else 'instruction-separators' if id.startswith('separator') else 'xsl-array' if id.startswith('array-') else 'xsl-switch' if id.startswith('switch-') else 'xsl-select'
 h.append(dict(v,entry='xslt-instructions/'+page+'.md',kind='current-draft-control' if v.get('ticket') else 'released-control',ticket=v.get('ticket')))
save('host-requests.json',h)
for name,program in [('c12-probe',Path('.guard-tmp/c11-probe/Program.cs').read_text(encoding='utf-8')),('c12-host','\n'.join(l for l in Path('.guard-tmp/c06-host/Program.cs').read_text(encoding='utf-8').splitlines() if not l.startswith('XsltFunctionLibrary.Register')))]:
 p=Path('.guard-tmp')/name;p.mkdir(exist_ok=True);(p/'Program.cs').write_text(program.replace('C11','C12').replace('C06','C12'),encoding='utf-8');shutil.copyfile('.guard-tmp/c11-probe/probe.csproj',p/'probe.csproj')
print(len(q),'API expressions;',len(h),'host requests')
