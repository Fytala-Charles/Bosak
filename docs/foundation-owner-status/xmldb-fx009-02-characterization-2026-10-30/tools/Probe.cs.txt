static void Require(bool value, string message) { if (!value) throw new InvalidOperationException("Qualification assertion: " + message); }
static ulong Target(ImmutableXmlSnapshot read) => Enumerable.Range(1, read.NodeCount).Select(i => read.GetNode((ulong)i)).First(n => n.Kind == StorageNodeKind.Element && n.LocalName == "n").Children.Single();
static List<IXdmNode> Nodes(XdmSequence values) { var result = new List<IXdmNode>(); foreach (var value in values) result.Add(value.NodeValue); return result; }
static XdmValue Single(XdmValue value) { if (value.Kind != XdmValueKind.Sequence) return value; foreach (var item in XdmSequence.FromSource(value.SequenceValue!)) return item; throw new InvalidOperationException("Empty diagnostic result."); }
static string Run(IXdmNode node, string expression) { try { var value = XPath31Expression.Compile(expression).Evaluate(new EvaluationContext().WithFocus(XdmValue.FromNode(node), 1, 1)); return value.Kind + ":" + value.ToString(); } catch (Exception error) { return "error:" + error.GetType().Name + ":" + error.Message; } }
static string Outcome(Action action) { try { action(); return "allowed"; } catch (Exception error) { return error.GetType().Name; } }
static bool Exclusive(string path) { try { using var handle = new FileStream(path, FileMode.Open, FileAccess.ReadWrite, FileShare.None); return true; } catch (IOException) { return false; } }

/// <summary>Owns a fresh diagnostic resource independently of borrowed result enumerators.</summary>
file sealed class ForeignSequence : IXdmSequence, IXdmSequenceEnumerator, IDisposable
{
    private readonly FileStream _stream;
    private int _position;
    internal ForeignSequence(string path) { Path = path; _stream = new(path, FileMode.CreateNew, FileAccess.ReadWrite, FileShare.None); }
    internal string Path { get; }
    internal int Fetches { get; private set; }
    internal int Closes { get; private set; }
    /// <inheritdoc />
    public bool TryGetLength(out int length) { length = 0; return false; }
    /// <inheritdoc />
    public IXdmSequenceEnumerator GetEnumerator() => this;
    /// <inheritdoc />
    public XdmValue Current => XdmValue.FromInteger(_position);
    /// <inheritdoc />
    public bool MoveNext() { Fetches++; return ++_position <= 2; }
    /// <inheritdoc />
    public void Dispose() { if (Closes != 0) return; Closes++; _stream.Dispose(); }
}
