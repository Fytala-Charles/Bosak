internal sealed record Member(string Name,string Identity,string Sha256,string Path,bool Readable);
internal sealed record Manifest(string Revision,Member[] Members);
internal sealed record Observation(Member[] Loaded,bool StableBinding,bool SupportedLayout);
internal sealed record Envelope(string RequestId,string RuntimeIdentity,string Profile,bool Entitled,bool Validated,bool ResourcesAdmitted);
internal sealed record Policy(bool SkipHash=false,bool FacadeOnly=false,bool AllowPartial=false,bool SkipWorkerIdentity=false,bool SkipProfile=false,bool SkipEntitlement=false)
{
    internal static readonly string[] Required=["Bosak.XPath.Core","Bosak.XPath.Parser","Bosak.XPath.Compiler","Bosak.XPath.Runtime","Bosak.XPath.Standard","Bosak.XPath.Providers","Bosak.XPath.Api","Bosak.XQuery","Bosak.Xslt"];
    internal bool Admit(Manifest approved,Observation actual)
    {
        if(!actual.StableBinding || !actual.SupportedLayout || (!Valid(approved.Members) || !Hex(approved.Revision,40)))return false;
        if(actual.Loaded.Any(m=>!m.Readable)||actual.Loaded.Select(m=>m.Name).Distinct(StringComparer.Ordinal).Count()!=actual.Loaded.Length)return false;
        if(!AllowPartial && !Valid(actual.Loaded))return false;
        var required=FacadeOnly ? approved.Members.Where(m=>m.Name=="Bosak.Xslt") : approved.Members.Where(m=>!AllowPartial || actual.Loaded.Any(a=>a.Name==m.Name));
        return required.All(expected=>actual.Loaded.Any(m=>m.Name==expected.Name && m.Identity==expected.Identity && (SkipHash||m.Sha256==expected.Sha256)));
    }
    private static bool Valid(Member[] members)=>members.Length==Required.Length && members.Select(m=>m.Name).ToHashSet(StringComparer.Ordinal).SetEquals(Required) && members.All(m=>m.Readable && !string.IsNullOrWhiteSpace(m.Identity) && m.Sha256.Length==64 && m.Sha256.All(c=>c is >= '0' and <= '9' or >= 'a' and <= 'f'));
    internal static string Digest(Manifest manifest)
    {
        if(!Valid(manifest.Members) || !Hex(manifest.Revision,40))throw new ArgumentException("Invalid approved closure");
        var canonical=JsonSerializer.Serialize(new {schema="prototype-manifest-v1",revision=manifest.Revision,members=manifest.Members.OrderBy(m=>m.Name,StringComparer.Ordinal).Select(m=>new {name=m.Name,identity=m.Identity,sha256=m.Sha256})});
        return Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(canonical))).ToLowerInvariant();
    }
    internal static string RuntimeIdentity(Manifest manifest)=>"prototype-runtime-v1:"+manifest.Revision+":"+Digest(manifest);
    private static bool Hex(string text,int length)=>text.Length==length && text.All(c=>c is >= '0' and <= '9' or >= 'a' and <= 'f');
    private static bool Runtime(string text) { var parts=text.Split(':'); return parts.Length==3 && parts[0]=="prototype-runtime-v1" && Hex(parts[1],40) && Hex(parts[2],64); }
    internal static bool Current(string retained,string verified,bool fresh)=>fresh && Runtime(retained) && retained==verified;
    internal bool Accept(Envelope parent,Envelope worker)=>!string.IsNullOrWhiteSpace(parent.RequestId) && !string.IsNullOrWhiteSpace(parent.Profile) && Runtime(parent.RuntimeIdentity) && Runtime(worker.RuntimeIdentity) && parent.RequestId==worker.RequestId && (SkipWorkerIdentity||parent.RuntimeIdentity==worker.RuntimeIdentity) && (SkipProfile||parent.Profile==worker.Profile) && (SkipEntitlement||parent.Entitled&&worker.Entitled) && parent.Validated&&worker.Validated && parent.ResourcesAdmitted&&worker.ResourcesAdmitted;
}
