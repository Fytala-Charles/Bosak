// Diagnostic policy guard; it is not wired into the real VM constructor paths.
static class RuleChecks {
 static bool MayReuse(NativeAdapter authority,SchemaAtomicItem source,XdmValue result,ISchemaComponentHandle declaredTarget,ISchemaComponentHandle actualTarget){
  var evidence=source.TypeEvidence;
  return evidence is not null&&authority.Owns(evidence.DeclaredType)&&authority.Owns(evidence.ActualType)&&authority.Owns(declaredTarget)&&authority.Owns(actualTarget)&&source.Value.Kind==result.Kind&&XdmValueEqualityComparer.Instance.Equals(source.Value,result)&&authority.IsSameComponent(evidence.DeclaredType,declaredTarget,default).Value&&authority.IsSameComponent(evidence.ActualType,actualTarget,default).Value;
 }
 public static void Run(Action<string,bool> check,NativeAdapter owner,SchemaAtomicItem cell,ISchemaComponentHandle other,ISchemaComponentHandle foreign){
  var type=cell.TypeEvidence.ActualType;
  check("R01-exact-noop",MayReuse(owner,cell,cell.Value,type,type));
  check("R02-changed-value",!MayReuse(owner,cell,XdmValue.FromDecimal(11m),type,type));
  check("R03-changed-kind-equal-number",!MayReuse(owner,cell,XdmValue.FromInteger(10),type,type));
  check("R04-foreign-target",!MayReuse(owner,cell,cell.Value,foreign,foreign));
  var differentBoundary=new SchemaAtomicItem(cell.Value,new NativeEvidence(other,type,cell.TypeEvidence.OriginalAssessment));
  check("R05-governing-boundary-mismatch",!MayReuse(owner,differentBoundary,cell.Value,type,type));
  var differentActual=new SchemaAtomicItem(cell.Value,new NativeEvidence(type,other,cell.TypeEvidence.OriginalAssessment));
  check("R06-actual-type-mismatch",!MayReuse(owner,differentActual,cell.Value,type,type));
 }
