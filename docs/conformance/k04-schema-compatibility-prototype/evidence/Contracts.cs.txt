/// <summary>Proposed isolated schema contract member; not a production API.</summary>
public sealed class SchemaOutcome<T>
{
    private readonly T _value;
    private static ImmutableArray<SchemaFinding> CheckFindings(ImmutableArray<SchemaFinding> findings)
    {
        if(findings.IsDefaultOrEmpty || findings.Any(f=>f is null || string.IsNullOrWhiteSpace(f.Reason) || string.IsNullOrWhiteSpace(f.Message)))throw new ArgumentException("Findings required.");
        return findings;
    }
    private SchemaOutcome(SchemaVerdict verdict,T value,ImmutableArray<SchemaFinding> findings) => (Verdict,_value,Findings)=(verdict,value,findings);
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    public SchemaVerdict Verdict { get; }
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    public T Value => Verdict==SchemaVerdict.Supported?_value:throw new InvalidOperationException("No supported value.");
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    public ImmutableArray<SchemaFinding> Findings { get; }
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="value">The proposed contract input.</param>
    /// <returns>The proposed contract result.</returns>
    public static SchemaOutcome<T> Supported(T value) => value is null ? throw new ArgumentNullException(nameof(value)) : new(SchemaVerdict.Supported,value,[]);
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <returns>The proposed contract result.</returns>
    public static SchemaOutcome<T> NotApplicable() => new(SchemaVerdict.NotApplicable,default!,[]);
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="findings">The proposed contract input.</param>
    /// <returns>The proposed contract result.</returns>
    public static SchemaOutcome<T> Invalid(ImmutableArray<SchemaFinding> findings) => new(SchemaVerdict.Invalid,default!,CheckFindings(findings));
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="findings">The proposed contract input.</param>
    /// <returns>The proposed contract result.</returns>
    public static SchemaOutcome<T> Unsupported(ImmutableArray<SchemaFinding> findings) => new(SchemaVerdict.Unsupported,default!,CheckFindings(findings));
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <returns>The proposed contract result.</returns>

/// <summary>Proposed isolated schema contract member; not a production API.</summary>
public sealed class SchemaIdentityIssuer
{
    private readonly SchemaSessionIdentity _identity;
    private readonly Dictionary<string,SchemaTypeIdentity> _types=new(StringComparer.Ordinal);
    private readonly Dictionary<string,SchemaDeclarationIdentity> _declarations=new(StringComparer.Ordinal);
    private bool _sealed;
    private SchemaIdentityIssuer(SchemaUniverseIdentity universe)
    {
        ArgumentNullException.ThrowIfNull(universe);ArgumentException.ThrowIfNullOrWhiteSpace(universe.ProviderModelVersion);
        if(!Enum.IsDefined(universe.SchemaLanguage) || universe.Fingerprint is not {Length:64} || universe.Fingerprint.Any(c=>!Uri.IsHexDigit(c)))throw new ArgumentException("Malformed universe identity.");
        _identity=new(universe with {Fingerprint=universe.Fingerprint.ToUpperInvariant()});
    }
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="universe">The complete provider-described universe.</param>
    /// <returns>The proposed contract result.</returns>
    public static SchemaIdentityIssuer Create(SchemaUniverseIdentity universe) => new(universe);
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    public SchemaSessionIdentity Identity => _identity;
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="key">The proposed contract input.</param>
    /// <param name="displayName">The proposed contract input.</param>
    /// <returns>The immutable identity interned within this issuer.</returns>
    public SchemaTypeIdentity InternType(string key,SchemaName? displayName)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(key);
        if(_types.TryGetValue(key,out var id))return id.DisplayName==displayName?id:throw new ArgumentException("Identity redefinition.");
        if(_sealed)throw new InvalidOperationException("Issuer sealed.");
        id=new(_identity,key,displayName);_types.Add(key,id);return id;
    }
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    /// <param name="key">The proposed contract input.</param>
    /// <param name="name">The proposed contract input.</param>
    /// <param name="scope">The proposed contract input.</param>
    /// <returns>The immutable scoped declaration identity.</returns>
    public SchemaDeclarationIdentity InternDeclaration(string key,SchemaName name,string scope)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(key);ArgumentException.ThrowIfNullOrWhiteSpace(scope);
        if(_declarations.TryGetValue(key,out var id))return id.Name==name&&id.Scope==scope?id:throw new ArgumentException("Identity redefinition.");
        if(_sealed)throw new InvalidOperationException("Issuer sealed.");
        id=new(_identity,key,name,scope);_declarations.Add(key,id);return id;
    }
    /// <summary>Proposed isolated schema contract member; not a production API.</summary>
    public void Seal() => _sealed=true;
}

{
    private SchemaSourceIdentity(string provider,string snapshotKey,string contentFingerprint)=>(Provider,SnapshotKey,ContentFingerprint)=(provider,snapshotKey,contentFingerprint);
    /// <summary>Creates a new source token; equal descriptive fields do not make tokens identical.</summary>
    /// <param name="provider">Host/source provider description.</param>
    /// <param name="snapshotKey">Host snapshot/generation key.</param>
    /// <param name="contentFingerprint">Retained source digest.</param>

    {

    }

    /// <summary>Gets the descriptive snapshot key; it does not authenticate a schema component.</summary>
