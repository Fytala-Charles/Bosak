using Bosak.Braid.Application;
using Bosak.Braid.Infrastructure;
/// <summary>Creates the actual inspector through public consumer APIs.</summary>
public static class Factory
{
    /// <summary>Creates an inspector with the supplied shared resource contract.</summary>
    /// <param name="reader">Host-owned resource reader.</param>
    /// <param name="maximumModules">Module budget.</param>
    /// <returns>The actual inspector implementing the shared host interface.</returns>
    public static ISourceInspector Create(IResourceReader? reader,int maximumModules) => new BosakSourceInspector(reader,maximumModules:maximumModules);
}
