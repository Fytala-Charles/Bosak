/// <summary>Neutral owned-unsupported versus successful authority outcome.</summary>
public enum SchemaResultKind
{
    /// <summary>A complete supported answer, including false or empty answers.</summary>
    Supported,
    /// <summary>A supported lookup found no component.</summary>
    NotFound,
    /// <summary>The authority cannot answer the owned capability.</summary>
    Unsupported,
    /// <summary>Owned simple-value construction proved the input invalid.</summary>
    Invalid
}
/// <summary>Explicit authority result; construction/invariants remain part of review.</summary>
/// <typeparam name="T">The supported payload type.</typeparam>
public abstract class SchemaResult<T>
{
    /// <summary>The outcome category.</summary>
    public abstract SchemaResultKind Kind { get; }
    /// <summary>The supported payload.</summary>
    public abstract T Value { get; }
    /// <summary>The optional provider diagnostic identifier.</summary>
    public abstract string? DiagnosticCode { get; }
}
/// <summary>Provider-owned authority operations; no issuer factories on the public host.</summary>
public interface ISchemaAuthority
{
    /// <summary>Checks actual membership, not the handle's own claimed Authority.</summary>
    /// <param name="component">The purported component.</param>
    /// <returns>True only for a component issued by this authority.</returns>
    bool Owns(ISchemaComponentHandle component);
    /// <summary>Resolves a type by name without treating that name as authority.</summary>
    /// <param name="name">The expanded lookup name.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>An owned handle, NotFound, or Unsupported.</returns>
    SchemaResult<ISchemaComponentHandle> ResolveType(SchemaExpandedName name, CancellationToken cancellationToken);
    /// <summary>Resolves a element declaration by name without treating that name as authority.</summary>
    /// <param name="name">The expanded lookup name.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>An owned handle, NotFound, or Unsupported.</returns>
    SchemaResult<ISchemaComponentHandle> ResolveElementDeclaration(SchemaExpandedName name, CancellationToken cancellationToken);
    /// <summary>Resolves a attribute declaration by name without treating that name as authority.</summary>
    /// <param name="name">The expanded lookup name.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>An owned handle, NotFound, or Unsupported.</returns>
    SchemaResult<ISchemaComponentHandle> ResolveAttributeDeclaration(SchemaExpandedName name, CancellationToken cancellationToken);

    /// <summary>Compares component identity using authority, without mandated handle interning.</summary>
    /// <param name="first">First owned component.</param>
    /// <param name="second">Second owned component.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>The authoritative identity comparison or unsupported outcome.</returns>
    SchemaResult<bool> IsSameComponent(ISchemaComponentHandle first, ISchemaComponentHandle second, CancellationToken cancellationToken);
    /// <summary>Queries an actual derivation relation.</summary>
    /// <param name="actual">Original candidate type.</param>
    /// <param name="expected">Expected governing type.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>A supported relation or explicit unsupported outcome.</returns>
    SchemaResult<bool> IsDerivedFrom(ISchemaComponentHandle actual, ISchemaComponentHandle expected, CancellationToken cancellationToken);
    /// <summary>Queries substitution affiliation without treating names as authority.</summary>
    /// <param name="member">Candidate declaration.</param>
    /// <param name="head">Expected head declaration.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>A supported relation or explicit unsupported outcome.</returns>
    SchemaResult<bool> IsAffiliatedWith(ISchemaComponentHandle member, ISchemaComponentHandle head, CancellationToken cancellationToken);
    /// <summary>Returns complete declared reachable components, without silent truncation.</summary>
    /// <param name="governingType">The governing original type.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>The complete closure or explicit unsupported outcome.</returns>
    SchemaResult<ImmutableArray<ISchemaComponentHandle>> GetComponentClosure(ISchemaComponentHandle governingType, CancellationToken cancellationToken);
}

/// <summary>Bound borrowed advisor; does not own caller snapshots or host operation control.</summary>
public interface ISchemaAssessmentAdvisor
{
    /// <summary>Exact compiled schema endpoint.</summary>
    ISchemaAssessor Assessor { get; }
    /// <summary>Exact environment captured at binding.</summary>
    IQuerySemanticEnvironment QueryEnvironment { get; }
    /// <summary>Verifies actual issuance for this bound advisor.</summary>
    /// <param name="origin">Original assessment to verify.</param>
    /// <returns>True only for original evidence issued within this endpoint binding.</returns>
    bool OwnsAssessment(ISchemaAssessmentOrigin origin);
    /// <summary>Verifies original node facts, not a caller object copying valid handles.</summary>
    /// <param name="facts">The purported original facts.</param>
    /// <returns>True only for immutable facts issued by this bound advisor.</returns>
    bool OwnsNodeFacts(ISchemaNodeFacts facts);

    /// <summary>Checks live borrowed advisor resources before an operation.</summary>
    /// <param name="cancellationToken">Operation cancellation.</param>
    void ThrowIfUnavailable(CancellationToken cancellationToken);
    /// <summary>Performs provider-owned assessment and retains original evidence.</summary>
    /// <param name="request">Validated mode, input, governing handles and episode.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>A Valid, Invalid or Unsupported result; cancellation throws.</returns>
    ISchemaAssessmentResult Assess(SchemaAssessmentRequest request, CancellationToken cancellationToken);
    /// <summary>Constructs a simple typed value with fresh target evidence, not inherited validation authority.</summary>
    /// <param name="input">Atomic/lexical input under the adopted constructor profile.</param>
    /// <param name="targetType">Owned simple target type.</param>
    /// <param name="episode">Constant schema episode.</param>
    /// <param name="cancellationToken">Operation cancellation.</param>
    /// <returns>Supported typed value, Invalid or Unsupported; no fabricated original assessment.</returns>
    SchemaResult<ISchemaTypedValue> ConstructSimpleValue(XdmValue input, ISchemaComponentHandle targetType, ISchemaAssessmentEpisode episode, CancellationToken cancellationToken);
}
