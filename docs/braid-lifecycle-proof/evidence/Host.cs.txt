internal sealed class BarrierReader:IResourceReader
{
    internal TaskCompletionSource Entered{get;}=new(TaskCreationOptions.RunContinuationsAsynchronously);internal TaskCompletionSource CallbackDone{get;}=new(TaskCreationOptions.RunContinuationsAsynchronously);internal TaskCompletionSource Release{get;}=new(TaskCreationOptions.RunContinuationsAsynchronously);internal TaskCompletionSource CancellationSeen{get;}=new(TaskCreationOptions.RunContinuationsAsynchronously);
    internal Func<Task>? Callback{get;set;}internal bool SawCancellation{get;private set;}
    /// <inheritdoc />
    public async Task<SourceDocument> ReadAsync(Uri resolvedUri,CancellationToken cancellationToken=default)
    {
        using var registration=cancellationToken.Register(()=>{SawCancellation=true;CancellationSeen.TrySetResult();});Entered.TrySetResult();if(Callback is not null)await Callback();CallbackDone.TrySetResult();
        // Deliberately retain the callback despite cancellation until the controlling barrier releases it.
        await Release.Task;cancellationToken.ThrowIfCancellationRequested();return new SourceDocument(resolvedUri,Encoding.UTF8.GetBytes("<xsl:stylesheet xmlns:xsl='http://www.w3.org/1999/XSL/Transform' version='3.0'/>"));
    }
}

internal sealed class Admission(string code):InvalidOperationException(code){internal string Code{get;}=code;}
internal sealed class Gate(Session session):IDisposable
{
    private static readonly AsyncLocal<Scope?> Current=new();
    private readonly object sync=new();private string state="Ready";private string reason="";private long sequence;private long activeId;private long nextId;
    private readonly TaskCompletionSource drained=new(TaskCreationOptions.RunContinuationsAsynchronously);
    internal List<object> Events{get;}=[];internal int ForwardedCalls{get;private set;}internal int LeasesCreated{get;private set;}internal int LeasesReleased{get;private set;}internal int ReleaseRequests{get;private set;}
    internal string State{get{lock(sync)return state;}}internal string Reason{get{lock(sync)return reason;}}internal bool Active{get{lock(sync)return activeId!=0;}}
    internal object Profile{get{lock(sync)return new{profile="lifecycle-readonly-experiment-v1",state,reason,canDispatch=state=="Ready"&&activeId==0,operations=new[]{"Inspect","RenameTarget","SameNameNoChange"}};}}
    private void Event(string name){Events.Add(new{sequence=++sequence,name,state,reason,activeId});}
    private void Ready(){if(state!="Ready")throw new Admission(reason=="ArtifactRefused"?"ArtifactRefused":"SessionClosed");}
    internal IDisposable Enter(CancellationToken token=default)
    {
        token.ThrowIfCancellationRequested();long id;
        lock(sync){Ready();if(activeId!=0){Event("busy-refused");throw new Admission("Busy");}activeId=id=++nextId;LeasesCreated++;Event("lease-reserved");}
        var lease=new Lease(this,id,Current.Value);Current.Value=new(this,id,token);
        try{Validate();return lease;}catch{lease.Dispose();throw;}
    }
    private void Validate()
    {
        var scope=Current.Value;if(scope is null||!ReferenceEquals(scope.Owner,this))throw new Admission("InvalidLease");scope.Token.ThrowIfCancellationRequested();
        lock(sync){Ready();if(activeId!=scope.Id)throw new Admission("InvalidLease");}
        // Verification I/O and all adapter/callback work occur outside the lifecycle ordering lock.
        try{session.Demand();}catch(InvalidOperationException){Close("ArtifactRefused");throw new Admission("ArtifactRefused");}
        lock(sync){Ready();if(activeId!=scope.Id)throw new Admission("InvalidLease");}
        scope.Token.ThrowIfCancellationRequested();
    }
    internal void Forward(){Validate();lock(sync){Ready();ForwardedCalls++;Event("raw-dispatch");}}
    internal void Publish(){Validate();lock(sync){Ready();Event("publication-admitted");}}
    internal T Read<T>(Func<T> read){lock(sync){Ready();return read();}}
    internal void Close(string cause="UserClose")
    {
        var release=false;lock(sync){if(state=="Ready"){state="Quiescing";reason=cause;Event("quiesce");}else Event("repeat-close:"+cause);if(state=="Quiescing"&&activeId==0){state="Closed";ReleaseRequests++;Event("release-requested");release=true;}}
        if(release)Release();
    }
    private void End(long id)
    {
        var release=false;lock(sync){if(activeId!=id)throw new InvalidOperationException("Lease accounting invariant");if(Current.Value?.Token.IsCancellationRequested==true)Event("cancelled-token-at-lease-end");activeId=0;LeasesReleased++;Event("lease-ended");if(state=="Quiescing"){state="Closed";ReleaseRequests++;Event("release-requested");release=true;}}
        if(release)Release();
    }
    private void Release(){try{session.Dispose();drained.TrySetResult();}catch(Exception error){drained.TrySetException(error);}}
    internal async Task<bool> DrainAsync(TimeSpan bound,CancellationToken token=default)
    {
        var scope=Current.Value;lock(sync){if(scope is not null&&ReferenceEquals(scope.Owner,this)&&scope.Id==activeId)throw new Admission("SelfDrain");if(state=="Ready")throw new Admission("NotClosing");}
        try{await drained.Task.WaitAsync(bound,token);return true;}catch(TimeoutException){return false;}
    }
    /// <summary>Stops new admission without waiting for an active callback or engine call.</summary>
    public void Dispose()=>Close();
    private sealed record Scope(Gate Owner,long Id,CancellationToken Token);
    private sealed class Lease(Gate owner,long id,Scope? prior):IDisposable
    {
        private int ended;
        /// <inheritdoc />
        public void Dispose(){if(Interlocked.Exchange(ref ended,1)==0){try{owner.End(id);}finally{Current.Value=prior;}}}
    }
}
internal sealed class InspectorFacade:ISourceInspector
{
    private readonly Gate gate;private readonly ISourceInspector raw;private readonly SourceAuthoringCapabilities? facts;
    internal InspectorFacade(Gate gate,ISourceInspector raw){this.gate=gate;this.raw=raw;facts=raw.Capabilities;}
    /// <inheritdoc />
    public SourceAuthoringCapabilities? Capabilities=>gate.Read(()=>facts);
    internal SourceAuthoringCapabilities? EngineFacts => facts;
    internal object AdmissionProfile => gate.Profile;
    /// <inheritdoc />
    public async Task<SourceInspectionResult> InspectAsync(SourceDocument source,CancellationToken cancellationToken=default)
    {
        try
        {
            using var lease=gate.Enter(cancellationToken);gate.Forward();var result=await raw.InspectAsync(source,cancellationToken);gate.Publish();
            return result with{Capabilities=result.Capabilities,ExpressionEditor=null,ValueOfInserter=null,ValueOfDeleter=null,TextDeleter=null,TextInserter=null,LiteralElementInserter=null,TextBodyReplacer=null,LiteralAttributeEditor=null,
                LiteralElementEditor=!result.Success||result.Issues.Count!=0||result.LiteralElementEditor is null?null:new ElementFacade(gate,result.LiteralElementEditor)};
        }
        catch(Admission refusal){return new(false,[],[new InspectionIssue(refusal.Code,source.Uri,"Diagnostic session admission refused.")]);}
    }
}
internal sealed class ElementFacade(Gate gate,ILiteralElementEditor raw):ILiteralElementEditor
{
    /// <inheritdoc />
    public async Task<LiteralElementTargetResult> GetTargetAsync(Uri moduleUri,int nodeId,LiteralElementOperation operation,CancellationToken cancellationToken=default)
    {
        try
        {
            using var lease=gate.Enter(cancellationToken);if(operation!=LiteralElementOperation.Rename)return new(null,new("UnsupportedProfile","Only read-only rename target discovery is admitted."));
            gate.Forward();var result=await raw.GetTargetAsync(moduleUri,nodeId,operation,cancellationToken);gate.Publish();
            if(result.Failure is not null)return new(null,result.Failure);
            if(result.Target is null||result.Target.Operation!=operation||result.Target.ModuleUri!=moduleUri||result.Target.NodeId!=nodeId)return new(null,new("InvalidResponse","Raw target did not match requested authority."));
            return new(new Target(this,gate,result.Target),null);
        }
        catch(Admission refusal){return new(null,new(refusal.Code,"Diagnostic session admission refused."));}
    }
    /// <inheritdoc />
    public async Task<LiteralElementResult> ProposeAsync(ILiteralElementTarget target,string? name,CancellationToken cancellationToken=default)
    {
        try
        {
            using var lease=gate.Enter(cancellationToken);
            if(target is not Target owned||!ReferenceEquals(owned.Owner,this))return Failure("ForeignTarget");
            if(name!=owned.Raw.LocalName)return Failure("UnsupportedProfile");
            gate.Forward();var result=await raw.ProposeAsync(owned.Raw,name,cancellationToken);gate.Publish();
            if(result.Failure is not null)return new(null,null,result.Failure);
            if(result.Candidate is not null||!ReferenceEquals(result.NoChangeTarget,owned.Raw)||result.Review is not null)return Failure("InvalidResponse");
            return new(null,owned,null);
        }
        catch(Admission refusal){return Failure(refusal.Code);}
    }
    private static LiteralElementResult Failure(string code)=>new(null,null,new(code,"Diagnostic NoChange-only port refused this operation."));
    private sealed class Target:ILiteralElementTarget

internal sealed class Session:IDisposable
{
    internal static readonly string[] Required=["Bosak.XPath.Core","Bosak.XPath.Parser","Bosak.XPath.Compiler","Bosak.XPath.Runtime","Bosak.XPath.Standard","Bosak.XPath.Providers","Bosak.XPath.Api","Bosak.XQuery","Bosak.Xslt","Bosak.Braid.Infrastructure"];
    private readonly Manifest manifest;private readonly Dictionary<string,byte[]> snapshots=new(StringComparer.Ordinal);private readonly GuardedContext context;private readonly byte[] plugin;private bool revoked;private bool closed;
    internal IReadOnlyList<Assembly> Shared{get;}=[typeof(ISourceInspector).Assembly,typeof(SourceDocument).Assembly];
    internal string Root{get;}internal string Identity{get;}internal List<Loaded> Loaded{get;}=[];
    internal Session(Manifest approved,string root,string operationPlugin)
    {
        manifest=approved;Root=Path.GetFullPath(root);Identity=Digest(approved);plugin=File.ReadAllBytes(operationPlugin);
        if(manifest.Contracts.Length!=2 || !manifest.Contracts.Select(c=>c.Name).ToHashSet(StringComparer.Ordinal).SetEquals(Shared.Select(a=>a.GetName().Name!)))throw new InvalidOperationException("Invalid shared contract approval");
        foreach(var shared in Shared)
        {
            var expected=manifest.Contracts.Single(c=>c.Name==shared.GetName().Name);
            if(Hash(File.ReadAllBytes(shared.Location))!=expected.Sha256 || File.Exists(Path.Combine(Root,expected.Name+".dll")))throw new InvalidOperationException("Shared contract mismatch/duplicate");
        }
        foreach(var member in manifest.Members)
        {
            var file=Path.Combine(Root,member.Name+".dll");if((File.GetAttributes(file)&FileAttributes.ReparsePoint)!=0)throw new InvalidOperationException("Unsupported artifact link");
            var bytes=File.ReadAllBytes(file);if(Hash(bytes)!=member.Sha256||Metadata(bytes).Identity!=member.Identity)throw new InvalidOperationException("Artifact mismatch: "+member.Name);snapshots.Add(member.Name,bytes);
        }
        context=new GuardedContext(this);
    }
    internal ISourceInspector Create(IResourceReader? reader,int limit)
    {
        Demand();var assembly=context.Assemblies.SingleOrDefault(x=>x.GetName().Name=="InspectorFactory")??context.LoadFromStream(new MemoryStream(plugin,false));
        return Program.Make(assembly,reader,limit);
    }
    internal void Demand()
    {
        if(closed)throw new InvalidOperationException("Session closed");
        if(revoked)throw new InvalidOperationException("Session revoked");
        try{foreach(var member in manifest.Members)if(Hash(File.ReadAllBytes(Path.Combine(Root,member.Name+".dll")))!=member.Sha256)throw new InvalidOperationException("Runtime artifact drift: "+member.Name);}
        catch(Exception error){revoked=true;throw new InvalidOperationException("Runtime admission refused",error);}
    }
    internal static string Digest(Manifest manifest)
    {
        if(manifest.Members.Length!=10||!manifest.Members.Select(m=>m.Name).ToHashSet(StringComparer.Ordinal).SetEquals(Required)||manifest.Revision.Length!=40||!manifest.Revision.All(Uri.IsHexDigit))throw new InvalidOperationException("Invalid approval manifest");
        return "experiment-runtime-v1:"+manifest.Revision+":"+Hash(Encoding.UTF8.GetBytes(JsonSerializer.Serialize(manifest.Members.OrderBy(m=>m.Name,StringComparer.Ordinal).Select(m=>new{m.Name,m.Identity,m.Sha256}))));
    }
    internal static string Hash(byte[] bytes)=>Convert.ToHexString(SHA256.HashData(bytes)).ToLowerInvariant();
    private static (string Identity,string Mvid) Metadata(byte[] bytes)
    {
        using var pe=new PEReader(new MemoryStream(bytes,false));var reader=pe.GetMetadataReader();var definition=reader.GetAssemblyDefinition();
        var name=new AssemblyName{ Name=reader.GetString(definition.Name),Version=definition.Version,CultureName=definition.Culture.IsNil?"":reader.GetString(definition.Culture)};
        if(!definition.PublicKey.IsNil)name.SetPublicKey(reader.GetBlobBytes(definition.PublicKey));else name.SetPublicKeyToken([]);
        return(name.FullName,reader.GetGuid(reader.GetModuleDefinition().Mvid).ToString());
    }
    /// <summary>Requests unload of this isolated context; actual garbage collection is not certified.</summary>
    public void Dispose(){if(!closed){closed=true;context.Unload();}}
    private sealed class GuardedContext(Session owner):AssemblyLoadContext(isCollectible:true)
    {
        /// <inheritdoc />
        protected override Assembly? Load(AssemblyName requested)
        {
            var shared=owner.Shared.SingleOrDefault(a=>a.GetName().Name==requested.Name);
            if(shared is not null){if(shared.FullName!=requested.FullName)throw new InvalidOperationException("Shared reference identity mismatch");return shared;}
            if(!owner.snapshots.TryGetValue(requested.Name!,out var bytes))
            {if(requested.Name!.StartsWith("Bosak.",StringComparison.Ordinal))throw new InvalidOperationException("Unapproved engine assembly");return null;}
            var member=owner.manifest.Members.Single(x=>x.Name==requested.Name);var expected=new AssemblyName(member.Identity);
            if(requested.Version>expected.Version||!requested.GetPublicKeyToken().AsSpan().SequenceEqual(expected.GetPublicKeyToken()))throw new InvalidOperationException("Unapproved assembly reference");
            var assembly=LoadFromStream(new MemoryStream(bytes,false));var metadata=Metadata(bytes);
            if(assembly.FullName!=metadata.Identity||assembly.ManifestModule.ModuleVersionId.ToString()!=metadata.Mvid)throw new InvalidOperationException("Loaded snapshot identity mismatch");
            owner.Loaded.Add(new(requested.Name!,assembly.FullName!,Hash(bytes),metadata.Mvid,Path.Combine(owner.Root,requested.Name+".dll"),assembly.Location));return assembly;
        }
    }
